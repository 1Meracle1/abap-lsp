CLASS lcl_counter DEFINITION.
  PUBLIC SECTION.
    METHODS next RETURNING VALUE(rv_value) TYPE i.
  PRIVATE SECTION.
    DATA mv_value TYPE i.
ENDCLASS.

CLASS lcl_counter IMPLEMENTATION.
  METHOD next.
    mv_value = mv_value + 1.
    IF mv_value > 10.
      mv_value = 0.
    ELSEIF mv_value = 5.
      WRITE mv_value.
    ELSE.
      WRITE 'ok'.
    ENDIF.
    rv_value = mv_value.
  ENDMETHOD.
ENDCLASS.
