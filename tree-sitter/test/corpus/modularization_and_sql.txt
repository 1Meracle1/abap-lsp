REPORT zmod.

FORM run USING iv_id TYPE string.
  SELECT SINGLE status
    FROM zorders
    INTO @DATA(lv_status)
    WHERE id = @iv_id.
  IF sy-subrc = 0.
    WRITE lv_status.
  ENDIF.
ENDFORM.

START-OF-SELECTION.
  PERFORM run USING '4711'.
